CREATE FUNCTION public.film_in_stock(p_film_id integer, p_store_id integer, OUT p_film_count integer) RETURNS SETOF integer
    LANGUAGE sql
    AS $_$
     SELECT inventory_id
     FROM inventory
     WHERE film_id = $1
     AND store_id = $2
     AND inventory_in_stock(inventory_id);
$_$;


1. Output Parameters + Row Sets: The PostgreSQL function does two things at once: it returns a result set of integers (RETURNS SETOF integer) and utilizes an OUT parameter (p_film_count). In SQL Server, a standard User-Defined Function (UDF) cannot return a table/set and use an output parameter simultaneously.

2. The Best Tool for the Job: To perfectly mimic this behavior in SQL Server, we use a Stored Procedure. Stored procedures natively support returning row sets while simultaneously populating @p_film_count as an OUTPUT parameter.
create procedure dbo.film_in_stock
  @p_film_id int,
  @p_store_id int,
  @p_film_count int output
As
BEGIN
    select inventory_id from dbo.inventory where film_id = @p_film_id and store_id = @p_store_id and dbo.inventory_in_stock(inventory_id) = 1

    select @p_film_count = count(inventory_id)  from dbo.inventory where film_id = @p_film_id and store_id = @p_store_id and dbo.inventory_in_stock(inventory_id) = 1
END;
GO

CREATE FUNCTION public.inventory_in_stock(p_inventory_id integer) RETURNS boolean
    LANGUAGE plpgsql
    AS $$
DECLARE
    v_rentals INTEGER;
    v_out     INTEGER;
BEGIN
    -- AN ITEM IS IN-STOCK IF THERE ARE EITHER NO ROWS IN THE rental TABLE
    -- FOR THE ITEM OR ALL ROWS HAVE return_date POPULATED

    SELECT count(*) INTO v_rentals
    FROM rental
    WHERE inventory_id = p_inventory_id;

    IF v_rentals = 0 THEN
      RETURN TRUE;
    END IF;

    SELECT COUNT(rental_id) INTO v_out
    FROM inventory LEFT JOIN rental USING(inventory_id)
    WHERE inventory.inventory_id = p_inventory_id
    AND rental.return_date IS NULL;

    IF v_out > 0 THEN
      RETURN FALSE;
    ELSE
      RETURN TRUE;
    END IF;
END $$;
